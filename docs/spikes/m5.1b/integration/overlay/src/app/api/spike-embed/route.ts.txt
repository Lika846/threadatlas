/** M5.1b integration spike route (temporary). Synthetic text only; no database. */
import { closeAll, getEmbedder, startBusy, status, SYNTHETIC, type Mode } from "@/lib/semantic/spike-host";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

const json = (body: unknown, code = 200) => Response.json(body, { status: code, headers: { "cache-control": "no-store" } });

export async function GET(request: Request): Promise<Response> {
  const q = new URL(request.url).searchParams;
  const op = q.get("op") ?? "status";
  const m = q.get("mode");
  const mode: Mode = m === "bundled" || m === "process" ? m : "source";
  const scope = q.get("scope") === "module" ? "module" : "global";
  try {
    if (op === "status") return json(status());
    if (op === "close") return json({ closeMs: await closeAll(), ...status() });
    const t = performance.now();
    const e = await getEmbedder(mode, scope);
    const startMs = Math.round(performance.now() - t);
    if (op === "start") return json({ startMs, info: e.info, ...status() });
    if (op === "embed") {
      const v = await e.embedPassages(SYNTHETIC);
      const qv = await e.embedQuery("garden budget");
      const all = [...v, qv];
      const buf = Buffer.concat(all.map((x) => Buffer.from(x.buffer, x.byteOffset, x.byteLength)));
      return json({ startMs, vectors: buf.toString("base64"), ...status() });
    }
    if (op === "busy") {
      startBusy(e, Number(q.get("n") ?? 3000));
      return json({ startMs, ...status() });
    }
    return json({ error: "unknown op" }, 400);
  } catch (err) {
    return json({ error: (err as { code?: string })?.code ?? "unknown" }, 500);
  }
}
